Agent-friendly install

<script src="https://cdn.cloudbtl.com/decklens.js"></script>
<script>
DeckLens.createTracker({
  projectId: "brand-proposal",
  deckId: "2026-q2-proposal",
  endpoint: "https://collector.cloudbtl.com/api/events"
}).start();
</script>
